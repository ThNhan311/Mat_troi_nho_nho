    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'FitCenter',
      theme: AppTheme.lightTheme.copyWith(
        appBarTheme: const AppBarTheme(
          systemOverlayStyle: SystemUiOverlayStyle.dark,
        ),
      ),
      darkTheme: AppTheme.darkTheme.copyWith(
        appBarTheme: const AppBarTheme(
          systemOverlayStyle: SystemUiOverlayStyle.light,
        ),
      ),
      themeMode: _themeMode,
      home: LoginPageWrapper(
        isDark: _themeMode == ThemeMode.dark,
        onToggleTheme: _toggleTheme,
      ),
    );

  }
}

class LoginPageWrapper extends StatelessWidget {
  final bool isDark;
  final ValueChanged<bool> onToggleTheme;

  const LoginPageWrapper({
    super.key,
    required this.isDark,
    required this.onToggleTheme,
  });

  void _openDemo(BuildContext context) {
    Navigator.of(context).pushReplacement(
      MaterialPageRoute<void>(
        builder: (_) => HomePage(
          isDark: isDark,
          onToggleTheme: onToggleTheme,
          onLogout: () {
            Navigator.of(context).pushAndRemoveUntil(
              MaterialPageRoute<void>(
                builder: (_) => LoginPageWrapper(
                  isDark: isDark,
                  onToggleTheme: onToggleTheme,
                ),
              ),
                  (route) => false,
            );
          },
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return LoginPage(
      onOpenDemo: () => _openDemo(context),
    );
  }
}

class WorkoutHistoryData {
  // Lá»‹ch sá»­ cĂ¡c buá»•i táº­p.
  static final List<Map<String, String>> history = [];

  // Tráº¡ng thĂ¡i hoĂ n thĂ nh cá»§a tá»«ng bĂ i táº­p.
  static final Map<String, bool> exerciseCompletion = {};

  static String _exerciseKey(String workoutTitle, String exerciseName) {
    return '$workoutTitle::$exerciseName';
  }

  static bool isExerciseCompleted(
      String workoutTitle,
      String exerciseName,
      ) {
    return exerciseCompletion[
    _exerciseKey(workoutTitle, exerciseName)
    ] ??
        false;
  }

  // LÆ°u tráº¡ng thĂ¡i ngay khi ngÆ°á»i dĂ¹ng tick hoáº·c bá» tick.
  static void setExerciseCompleted({
    required String workoutTitle,
    required String exerciseName,
    required bool completed,
    required int completedCount,
    required int totalCount,
    required String duration,
  }) {
    exerciseCompletion[
    _exerciseKey(workoutTitle, exerciseName)
    ] = completed;

    // ChÆ°a hoĂ n thĂ nh bĂ i nĂ o thĂ¬ khĂ´ng ghi nháº­n buá»•i táº­p.
    if (completedCount == 0) {
      history.removeWhere(
            (item) => item['title'] == workoutTitle,
      );
      return;
    }

    final now = DateTime.now();
    final date =
        '${now.day.toString().padLeft(2, '0')}/'
        '${now.month.toString().padLeft(2, '0')}/'
        '${now.year}';

    final existingIndex = history.indexWhere(
          (item) => item['title'] == workoutTitle,
    );

    final record = <String, String>{
      'date': date,
      'title': workoutTitle,
      'completed': '$completedCount/$totalCount bĂ i táº­p',
      'duration': duration,
    };

    if (existingIndex >= 0) {
      history[existingIndex] = record;
    } else {
      history.insert(0, record);
    }
  }
}

  final List<Map<String, dynamic>> renewalHistory = [];

  Map<String, dynamic>? currentMembership;

      SnackBar(
        content: Text(message),
        behavior: SnackBarBehavior.floating,
      ),
    );
  }


  void _handlePaymentSuccess(Map<String, dynamic> payment) {
    final now = DateTime.now();

    // Láº¥y ngĂ y háº¿t háº¡n hiá»‡n táº¡i, náº¿u cĂ³.
    final oldExpiryText = currentMembership?['expiryDate']?.toString();
    final oldExpiry = oldExpiryText == null
        ? null
        : DateTime.tryParse(oldExpiryText);

    // Náº¿u gĂ³i cÅ© cĂ²n háº¡n thĂ¬ ná»‘i tiáº¿p tá»« ngĂ y háº¿t háº¡n cÅ©.

    final startDate = oldExpiry != null && oldExpiry.isAfter(now)
        ? oldExpiry
        : now;

    final duration = payment['duration']?.toString() ?? '';
    final days = duration.contains('180')
        ? 180
        : duration.contains('90')
        ? 90
        : 30;

    final expiryDate = startDate.add(Duration(days: days));

    final updatedPayment = {
      ...payment,
      'startDate': startDate.toIso8601String(),
      'expiryDate': expiryDate.toIso8601String(),
    };

      renewalHistory.insert(0, updatedPayment);

      currentMembership = {
        'planName': payment['planName'],
        'duration': duration,
        'price': payment['price'],
        'paidAt': payment['paidAt'],
        'startDate': startDate.toIso8601String(),
        'expiryDate': expiryDate.toIso8601String(),
      };
    });

  String _formatPaymentDate(String? value) {
    final date = value == null ? null : DateTime.tryParse(value);

    if (date == null) return '--/--/----';

    return '${date.day.toString().padLeft(2, '0')}/'
        '${date.month.toString().padLeft(2, '0')}/'
        '${date.year}';
  }

  String _formatPaymentPrice(dynamic value) {
    final price = (value as num?)?.round();

    if (price == null) return '0';

    final digits = price.toString();
    final result = StringBuffer();

    for (int i = 0; i < digits.length; i++) {
      if (i > 0 && (digits.length - i) % 3 == 0) {
        result.write('.');
      }
      result.write(digits[i]);
    }

    return result.toString();
  }

import 'package:http/http.dart' as http;

class BookingService {
  static const String baseUrl = 'http://10.0.2.2:8080/api/v1';

  Future<Map<String, dynamic>> createBooking({
    required String ptId,
    required String slotId,
    required String token,
  }) async {
    final uri = Uri.parse('$baseUrl/bookings');

    final response = await http.post(
      uri,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $token',
      },
      body: jsonEncode({
        'ptId': ptId,
        'slotId': slotId,
      }),
    );

    final body = response.body.isNotEmpty
        ? jsonDecode(response.body) as Map<String, dynamic>
        : <String, dynamic>{};

    if (response.statusCode == 201) {
      return body;
    }

    if (response.statusCode == 409) {
      throw Exception(
        body['message'] ??
            'Khung giá» nĂ y Ä‘Ă£ Ä‘Æ°á»£c ngÆ°á»i khĂ¡c Ä‘áº·t.',
      );
    }

    if (response.statusCode == 401) {
      throw Exception(
        'PhiĂªn Ä‘Äƒng nháº­p khĂ´ng há»£p lá»‡. Vui lĂ²ng Ä‘Äƒng nháº­p láº¡i.',
      );
    }

    if (response.statusCode == 403) {
      throw Exception(
        body['message'] ??
            'Báº¡n khĂ´ng cĂ³ quyá»n thá»±c hiá»‡n thao tĂ¡c nĂ y.',
      );
    }

    throw Exception(
      body['message'] ??
          'KhĂ´ng thá»ƒ Ä‘áº·t lá»‹ch. MĂ£ lá»—i: ${response.statusCode}',
    );
  }

  Future<List<Map<String, dynamic>>> getMyBookings({
    required String token,
  }) async {
    final uri = Uri.parse('$baseUrl/bookings/me');

    final response = await http.get(
      uri,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $token',
      },
    );

    if (response.statusCode != 200) {
      throw Exception(
        'KhĂ´ng thá»ƒ táº£i lá»‹ch sá»­ Ä‘áº·t lá»‹ch.',
      );
    }

    final decoded = jsonDecode(response.body);

    if (decoded is List) {
      return decoded
          .map(
            (item) => Map<String, dynamic>.from(
          item as Map,
        ),
      )
          .toList();
    }

    return [];
  }

  Future<Map<String, dynamic>> cancelBooking({
    required String bookingId,
    required String token,
  }) async {
    final uri = Uri.parse(
      '$baseUrl/bookings/$bookingId/cancel',
    );

    final response = await http.post(
      uri,
      headers: {

    );
